Use the [inventory](nho-proceedings-inventory.tsv) to name each accessible original `nho-<meeting>.pdf` in the capture directory. The upstream PyMuPDF CLI extracts a UTF-8 text file with form-feed page breaks, and the Free Pascal reader checks the original PDF and extracted page count before matching terms.

```powershell
$experiment = 'experiments/1000_see_what_sticks_food'
$followup = "$experiment/followup"
$capture = "$experiment/build/nho-proceedings"
$bin = "$experiment/build/followup-3.3.1-i386-win32/bin"
$inventory = "$followup/nho-proceedings-inventory.tsv"
Import-Csv $inventory -Delimiter "`t" | Where-Object sha256 | ForEach-Object {
    $stem = "$capture/nho-$($_.meeting)"
    python -m pymupdf gettext -mode simple -output "$stem.formfeed" "$stem.pdf"
    if ($LASTEXITCODE) { throw "Text extraction failed for $stem" }
}
& "$bin/foodexperiment.jsdr.exe" $inventory "$followup/jsdr-topic-vocabulary.tsv" $capture $experiment "$followup/jsdr-claim-vocabulary.tsv" NHO
if ($LASTEXITCODE) { throw "National Hospital Organization screen failed" }
& "$bin/foodexperiment.crossref.exe" audit-topics "$followup/nho-proceedings-review-keys.tsv" "$capture/context-audit.tsv" "$followup/nho-proceedings-context-reviews.tsv"
if ($LASTEXITCODE) { throw "National Hospital Organization contextual audit failed" }
```
